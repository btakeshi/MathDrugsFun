function simplifyRadical(n) {
  let k = 1, m = n;
  for (let i = Math.floor(Math.sqrt(n)); i >= 2; i--) {
    if (n % (i * i) === 0) { k = i; m = n / (i * i); break; }
  }
  if (m === 1) return `${k}`;
  return k === 1 ? `\\sqrt{${m}}` : `${k}\\sqrt{${m}}`;
}

ToolEngine.init(function() {
  const a = parseFloat(document.getElementById('val-a').value);
  const b = parseFloat(document.getElementById('val-b').value);
  if (!a || !b) return null;
  const c2 = a*a + b*b;
  return {
    problem: `底辺 $a=${a}$, 高さ $b=${b}$ の直角三角形の斜辺 $c$ を求めなさい。`,
    steps: [
      { title: "公式", content: "三平方の定理 $a^2 + b^2 = c^2$ を使います。" },
      { title: "代入", content: "$c^2 = " + a + "^2 + " + b + "^2 = " + c2 + "$" },
      { title: "結果", content: `答えは $c = ${simplifyRadical(c2)}$ です。` }
    ]
  };
});
</script>
